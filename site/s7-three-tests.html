The code

Try the page first: type 50 in the price box and click Search. Then read the code that did it.

// The rule, on its own: one question, one answer.
function isAffordable(price, maxPrice) {
  return maxPrice === '' || Number(price) <= Number(maxPrice);
}

// The controller: reads the page, applies the rule to each part, shows or hides it.
function showParts() {
  const query = document.getElementById('part-number').value.trim().toUpperCase();
  const maxPrice = document.getElementById('max-price').value.trim();
  document.querySelectorAll('.catalog-part').forEach(card => {
    const matches = card.dataset.part.startsWith(query);
    card.hidden = !(matches && isAffordable(card.dataset.price, maxPrice));
  });
}
